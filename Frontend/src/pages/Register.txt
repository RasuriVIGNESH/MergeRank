/* ═══════════════════════════════════════════════════════════════════════ */
const Register = () => {
    const navigate = useNavigate();
    const [dark, setDark] = useState(false);
    const [showPassword, setShowPassword] = useState(false);
    const [showConfirm, setShowConfirm] = useState(false);
    const [formData, setFormData] = useState({
        firstName: '', lastName: '', email: '',
        college: '', gradYear: '', branch: '', batchCode: '',
        password: '', confirmPassword: '',
    });
    const [error, setError] = useState('');
    const [loading, setLoading] = useState(false);

    const D = dark;

    const handleChange = e => setFormData({ ...formData, [e.target.name]: e.target.value });

    const calcStrength = (p) => {
        if (!p) return 0;
        let s = 0;
        if (p.length > 8) s++;
        if (/[A-Z]/.test(p)) s++;
        if (/[0-9]/.test(p)) s++;
        if (/[^A-Za-z0-9]/.test(p)) s++;
        return s;
    };
    const strength = calcStrength(formData.password);
    const strengthMeta = [
        { color: '#ef4444', label: 'Weak' },
        { color: '#f97316', label: 'Fair' },
        { color: '#eab308', label: 'Good' },
        { color: '#22c55e', label: 'Strong' },
    ];

    const handleSubmit = async (e) => {
        e.preventDefault();
        setError('');
        if (formData.password !== formData.confirmPassword) { setError('Passwords do not match.'); return; }
        setLoading(true);
        try {
            await authService.register({
                name: `${formData.firstName} ${formData.lastName}`,
                email: formData.email, password: formData.password,
                gradYear: formData.gradYear, branch: formData.branch,
                batch: formData.batchCode, college: formData.college,
            });
            navigate('/onboarding');
        } catch (err) {
            setError(err.response?.data?.message || 'Registration failed. Please try again.');
        } finally {
            setLoading(false);
        }
    };

    /* ── Colours ── */
    const rootBg = D ? '#080c14' : '#f8faff';
    const cardBg = D ? '#111827' : '#ffffff';
    const cardBdr = D ? 'rgba(255,255,255,0.08)' : '#e5e7eb';
    const textMain = D ? '#f0f4ff' : '#0f172a';
    const textSub = D ? '#94a3b8' : '#6b7280';
    const linkClr = D ? '#818cf8' : '#2563eb';
    const pillBg = D ? 'rgba(99,102,241,0.15)' : '#eff6ff';
    const pillClr = D ? '#a5b4fc' : '#2563eb';
    const pillBdr = D ? 'rgba(99,102,241,0.3)' : '#bfdbfe';
    const toggleBg = D ? '#1e2a3d' : '#f3f4f6';
    const toggleBdr = D ? 'rgba(255,255,255,0.1)' : '#e5e7eb';
    const inputCls = `reg-input ${D ? 'reg-input-dark' : 'reg-input-light'}`;

    return (
        <div className="reg-root" style={{ minHeight: '100vh', display: 'flex', background: rootBg, transition: 'background 0.3s' }}>
            <Styles />

            {/* ── LEFT PANEL ─────────────────────────────────────────────── */}
            <div style={{ width: '42%', minHeight: '100vh', background: D ? '#0d1220' : '#0f172a', position: 'relative', display: 'flex', flexDirection: 'column', overflow: 'hidden', flexShrink: 0 }} className="hidden lg:flex">

                {/* Glow blobs */}
                <div style={{ position: 'absolute', top: '-10%', left: '-10%', width: 360, height: 360, borderRadius: '50%', background: 'rgba(99,102,241,0.2)', filter: 'blur(80px)', pointerEvents: 'none' }} />
                <div style={{ position: 'absolute', bottom: '-15%', right: '-15%', width: 420, height: 420, borderRadius: '50%', background: 'rgba(14,165,233,0.12)', filter: 'blur(90px)', pointerEvents: 'none' }} />
                <div className="dot-pattern" style={{ position: 'absolute', inset: 0, opacity: 0.22, pointerEvents: 'none' }} />

                {/* Brand + copy */}
                <div style={{ padding: '44px 48px 0', position: 'relative', zIndex: 1 }}>
                    <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 44 }}>
                        <div style={{ width: 34, height: 34, borderRadius: 9, background: 'linear-gradient(135deg,#2563eb,#6366f1)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                            <GitMerge style={{ width: 16, height: 16, color: '#fff' }} />
                        </div>
                        <span style={{ fontSize: 17, fontWeight: 700, color: '#f0f4ff', letterSpacing: '-0.02em' }}>MergeRank</span>
                    </div>

                    <div style={{ marginBottom: 18 }}>
                        <span style={{ fontSize: 11, fontWeight: 700, textTransform: 'uppercase', letterSpacing: '0.1em', color: '#6366f1', background: 'rgba(99,102,241,0.15)', padding: '4px 12px', borderRadius: 20, border: '1px solid rgba(99,102,241,0.3)' }}>
                            Join the platform
                        </span>
                    </div>

                    <h2 className="reg-serif" style={{ fontSize: 'clamp(26px,2.5vw,38px)', lineHeight: 1.12, fontWeight: 400, color: '#f0f4ff', marginBottom: 14 }}>
                        Track your growth.<br />
                        <span style={{ color: '#818cf8', fontStyle: 'italic' }}>Prove your potential.</span>
                    </h2>
                    <p style={{ fontSize: 13, lineHeight: 1.7, color: '#64748b', maxWidth: 300, marginBottom: 36 }}>
                        Unify your competitive programming profiles into one powerful dashboard. Stand out to mentors.
                    </p>

                    {/* Perks */}
                    <div style={{ display: 'flex', flexDirection: 'column', gap: 10 }}>
                        {PERKS.map((p, i) => (
                            <div key={i} style={{ display: 'flex', alignItems: 'flex-start', gap: 10 }}>
                                <CheckCircle2 style={{ width: 15, height: 15, color: '#22c55e', marginTop: 1, flexShrink: 0 }} />
                                <span style={{ fontSize: 13, color: '#94a3b8', lineHeight: 1.5 }}>{p}</span>
                            </div>
                        ))}
                    </div>
                </div>

                {/* Orbit ring */}
                <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', position: 'relative', zIndex: 1, minHeight: 260 }}>
                    <div style={{ position: 'relative', width: 240, height: 240 }}>
                        {/* Ring */}
                        <div style={{ position: 'absolute', top: '50%', left: '50%', transform: 'translate(-50%,-50%)', width: 216, height: 216, borderRadius: '50%', border: '1px dashed rgba(99,102,241,0.25)' }} />

                        {/* Centre MR hub */}
                        <div style={{ position: 'absolute', top: '50%', left: '50%', transform: 'translate(-50%,-50%)', width: 60, height: 60, borderRadius: 16, background: 'linear-gradient(135deg,#2563eb,#6366f1)', display: 'flex', alignItems: 'center', justifyContent: 'center', boxShadow: '0 0 40px rgba(99,102,241,0.45)', zIndex: 3 }}>
                            <span style={{ fontSize: 15, fontWeight: 800, color: '#fff' }}>MR</span>
                        </div>

                        {/* Orbiting nodes */}
                        {PLATFORMS.map((p, i) => (
                            <div key={i} className={`orbit-node-${i}`} style={{ position: 'absolute', top: '50%', left: '50%', zIndex: 2 }}>
                                <div style={{ width: 42, height: 42, borderRadius: '50%', background: 'rgba(255,255,255,0.06)', border: '1px solid rgba(255,255,255,0.12)', display: 'flex', alignItems: 'center', justifyContent: 'center', backdropFilter: 'blur(6px)' }}>
                                    <img src={p.logo} alt={p.name} style={{ width: 20, height: 20, objectFit: 'contain' }} />
                                </div>
                            </div>
                        ))}
                    </div>
                </div>

                {/* Bottom bar */}
                <div style={{ padding: '20px 48px', borderTop: '1px solid rgba(255,255,255,0.06)', position: 'relative', zIndex: 1 }}>
                    <div style={{ display: 'flex', gap: 6, flexWrap: 'wrap' }}>
                        {PLATFORMS.map(p => (
                            <span key={p.name} style={{ fontSize: 11, color: '#4b5563', background: 'rgba(255,255,255,0.04)', padding: '3px 8px', borderRadius: 6, border: '1px solid rgba(255,255,255,0.06)' }}>{p.name}</span>
                        ))}
                    </div>
                </div>
            </div>

            {/* ── RIGHT PANEL ────────────────────────────────────────────── */}
            <div style={{ flex: 1, display: 'flex', flexDirection: 'column', minHeight: '100vh', background: rootBg, transition: 'background 0.3s', overflowY: 'auto' }}>

                {/* Top bar */}
                <div style={{ flexShrink: 0, display: 'flex', justifyContent: 'space-between', alignItems: 'center', padding: '14px 32px', borderBottom: `1px solid ${D ? 'rgba(255,255,255,0.06)' : '#f0f0f0'}` }}>
                    {/* Mobile logo */}
                    <div style={{ display: 'flex', alignItems: 'center', gap: 8 }} className="lg:hidden">
                        <div style={{ width: 26, height: 26, borderRadius: 7, background: 'linear-gradient(135deg,#2563eb,#6366f1)', display: 'flex', alignItems: 'center', justifyContent: 'center' }}>
                            <GitMerge style={{ width: 13, height: 13, color: '#fff' }} />
                        </div>
                        <span style={{ fontSize: 14, fontWeight: 700, color: textMain }}>MergeRank</span>
                    </div>

                    <div style={{ display: 'flex', alignItems: 'center', gap: 14, marginLeft: 'auto' }}>
                        <span style={{ fontSize: 13, color: textSub }}>
                            Already have an account?{' '}
                            <button onClick={() => navigate('/login')} style={{ background: 'none', border: 'none', cursor: 'pointer', fontSize: 13, fontWeight: 600, color: linkClr, padding: 0 }}>
                                Log in
                            </button>
                        </span>
                        <button onClick={() => setDark(d => !d)}
                            style={{ width: 32, height: 32, borderRadius: 8, border: `1px solid ${toggleBdr}`, background: toggleBg, display: 'flex', alignItems: 'center', justifyContent: 'center', cursor: 'pointer', color: D ? '#fbbf24' : '#6b7280', transition: 'all 0.2s' }}
                            aria-label="Toggle theme">
                            {D ? <Sun style={{ width: 14, height: 14 }} /> : <Moon style={{ width: 14, height: 14 }} />}
                        </button>
                    </div>
                </div>

                {/* Form */}
                <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', padding: '32px 24px' }}>
                    <div style={{ width: '100%', maxWidth: 460 }}>

                        {/* Heading */}
                        {/* <div className="anim-fade-up" style={{ marginBottom: 26 }}>
                            <h1 className="reg-serif" style={{ fontSize: 34, fontWeight: 400, lineHeight: 1.12, color: textMain }}>
                                Get started<span className="gradient-text">.</span>
                            </h1>
                            <p style={{ fontSize: 13, color: textSub, lineHeight: 1.6, marginTop: 8 }}>
                                Join your batch and start tracking your competitive programming journey.
                            </p>
                        </div> */}

                        {/* Error */}
                        {error && (
                            <div className="anim-fade-up" style={{ marginBottom: 16, padding: '11px 14px', background: D ? 'rgba(239,68,68,0.1)' : '#fef2f2', border: `1px solid ${D ? 'rgba(239,68,68,0.25)' : '#fecaca'}`, borderRadius: 12, display: 'flex', alignItems: 'center', gap: 8, fontSize: 13, color: D ? '#fca5a5' : '#dc2626' }}>
                                <AlertCircle style={{ width: 14, height: 14, flexShrink: 0 }} />
                                {error}
                            </div>
                        )}

                        {/* Card */}
                        <div className="anim-fade-up d1" style={{ background: cardBg, border: `1px solid ${cardBdr}`, borderRadius: 20, padding: '26px 28px', boxShadow: D ? '0 20px 60px rgba(0,0,0,0.4)' : '0 8px 40px rgba(0,0,0,0.07)' }}>

                            <form onSubmit={handleSubmit} style={{ display: 'flex', flexDirection: 'column', gap: 14 }}>

                                {/* ── Section: Personal ── */}
                                <div style={{ fontSize: 10, fontWeight: 800, textTransform: 'uppercase', letterSpacing: '0.12em', color: D ? '#374151' : '#d1d5db', paddingBottom: 4, borderBottom: `1px solid ${D ? 'rgba(255,255,255,0.05)' : '#f5f5f5'}` }}>
                                    Personal Info
                                </div>

                                {/* Name row */}
                                <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 12 }}>
                                    <Field label="First Name" dark={D}>
                                        <input type="text" name="firstName" required value={formData.firstName} onChange={handleChange} placeholder="Alex" className={inputCls} />
                                    </Field>
                                    <Field label="Last Name" dark={D}>
                                        <input type="text" name="lastName" required value={formData.lastName} onChange={handleChange} placeholder="Johnson" className={inputCls} />
                                    </Field>
                                </div>

                                <Field label="College Email" dark={D}>
                                    <input type="email" name="email" required value={formData.email} onChange={handleChange} placeholder="you@college.edu" className={inputCls} />
                                </Field>

                                {/* ── Section: Academic ── */}
                                <div style={{ fontSize: 10, fontWeight: 800, textTransform: 'uppercase', letterSpacing: '0.12em', color: D ? '#374151' : '#d1d5db', paddingBottom: 4, borderBottom: `1px solid ${D ? 'rgba(255,255,255,0.05)' : '#f5f5f5'}`, marginTop: 4 }}>
                                    Academic Details
                                </div>

                                <Field label="College" dark={D}>
                                    <select name="college" required value={formData.college} onChange={handleChange} className={`${inputCls} reg-select`} style={{ paddingLeft: 12 }}>
                                        <option value="" disabled>Select College</option>
                                        {['Woxsen'].map(c => <option key={c} value={c}>{c}</option>)}
                                    </select>
                                </Field>

                                <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 12 }}>
                                    <Field label="Graduation Year" dark={D}>
                                        <select name="gradYear" required value={formData.gradYear} onChange={handleChange} className={`${inputCls} reg-select`} style={{ paddingLeft: 12 }}>
                                            <option value="" disabled>Year</option>
                                            {[2025, 2026, 2027, 2028, 2029].map(y => <option key={y} value={y}>{y}</option>)}
                                        </select>
                                    </Field>
                                    <Field label="Branch" dark={D}>
                                        <select name="branch" required value={formData.branch} onChange={handleChange} className={`${inputCls} reg-select`} style={{ paddingLeft: 12 }}>
                                            <option value="" disabled>Branch</option>
                                            {[['CS', 'Computer Science'], ['IT', 'Info. Tech'], ['ECE', 'Electronics'], ['EE', 'Electrical'], ['ME', 'Mechanical']].map(([v, l]) => (
                                                <option key={v} value={v}>{l}</option>
                                            ))}
                                        </select>
                                    </Field>
                                </div>

                                {/* ── Section: Security ── */}
                                <div style={{ fontSize: 10, fontWeight: 800, textTransform: 'uppercase', letterSpacing: '0.12em', color: D ? '#374151' : '#d1d5db', paddingBottom: 4, borderBottom: `1px solid ${D ? 'rgba(255,255,255,0.05)' : '#f5f5f5'}`, marginTop: 4 }}>
                                    Security
                                </div>

                                <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 12 }}>
                                    {/* Password */}
                                    <Field label="Password" dark={D}>
                                        <div style={{ position: 'relative' }}>
                                            <input
                                                type={showPassword ? 'text' : 'password'}
                                                name="password" required
                                                value={formData.password} onChange={handleChange}
                                                placeholder="Min 8 chars"
                                                className={inputCls} style={{ paddingRight: 36 }}
                                            />
                                            <button type="button" onClick={() => setShowPassword(v => !v)}
                                                style={{ position: 'absolute', right: 11, top: '50%', transform: 'translateY(-50%)', background: 'none', border: 'none', cursor: 'pointer', color: D ? '#4b5563' : '#9ca3af', display: 'flex', transition: 'color 0.15s' }}
                                                onMouseEnter={e => e.currentTarget.style.color = '#2563eb'}
                                                onMouseLeave={e => e.currentTarget.style.color = D ? '#4b5563' : '#9ca3af'}>
                                                {showPassword ? <EyeOff style={{ width: 13, height: 13 }} /> : <Eye style={{ width: 13, height: 13 }} />}
                                            </button>
                                        </div>
                                        {/* Strength bar */}
                                        {formData.password.length > 0 && (
                                            <div style={{ marginTop: 6 }}>
                                                <div style={{ display: 'flex', gap: 3, marginBottom: 3 }}>
                                                    {[0, 1, 2, 3].map(i => (
                                                        <div key={i} className="strength-bar" style={{ background: i < strength ? strengthMeta[strength - 1].color : (D ? '#1e2a3d' : '#e5e7eb') }} />
                                                    ))}
                                                </div>
                                                {strength > 0 && <span style={{ fontSize: 10, fontWeight: 600, color: strengthMeta[strength - 1].color }}>{strengthMeta[strength - 1].label}</span>}
                                            </div>
                                        )}
                                    </Field>

                                    {/* Confirm Password */}
                                    <Field label="Confirm Password" dark={D}>
                                        <div style={{ position: 'relative' }}>
                                            <input
                                                type={showConfirm ? 'text' : 'password'}
                                                name="confirmPassword" required
                                                value={formData.confirmPassword} onChange={handleChange}
                                                placeholder="Re-enter"
                                                className={`${inputCls}${formData.confirmPassword && formData.password !== formData.confirmPassword ? ' reg-input-error' : ''}`}
                                                style={{ paddingRight: 36 }}
                                            />
                                            <button type="button" onClick={() => setShowConfirm(v => !v)}
                                                style={{ position: 'absolute', right: 11, top: '50%', transform: 'translateY(-50%)', background: 'none', border: 'none', cursor: 'pointer', color: D ? '#4b5563' : '#9ca3af', display: 'flex', transition: 'color 0.15s' }}
                                                onMouseEnter={e => e.currentTarget.style.color = '#2563eb'}
                                                onMouseLeave={e => e.currentTarget.style.color = D ? '#4b5563' : '#9ca3af'}>
                                                {showConfirm ? <EyeOff style={{ width: 13, height: 13 }} /> : <Eye style={{ width: 13, height: 13 }} />}
                                            </button>
                                        </div>
                                        {formData.confirmPassword && formData.password !== formData.confirmPassword && (
                                            <p style={{ fontSize: 10, color: '#ef4444', marginTop: 4 }}>Passwords don't match</p>
                                        )}
                                        {formData.confirmPassword && formData.password === formData.confirmPassword && formData.confirmPassword.length > 0 && (
                                            <p style={{ fontSize: 10, color: '#22c55e', marginTop: 4, display: 'flex', alignItems: 'center', gap: 4 }}>
                                                <CheckCircle2 style={{ width: 10, height: 10 }} /> Passwords match
                                            </p>
                                        )}
                                    </Field>
                                </div>

                                {/* Submit */}
                                <button type="submit" disabled={loading} className="shimmer-btn"
                                    style={{ width: '100%', height: 44, borderRadius: 11, fontSize: 14, fontWeight: 600, color: '#fff', display: 'flex', alignItems: 'center', justifyContent: 'center', gap: 8, marginTop: 6 }}>
                                    {loading ? 'Creating Account…' : <><span>Create Account</span><ArrowRight style={{ width: 15, height: 15 }} /></>}
                                </button>

                                {/* Terms */}
                                <p style={{ textAlign: 'center', fontSize: 11, color: D ? '#374151' : '#9ca3af', lineHeight: 1.6 }}>
                                    By registering, you agree to our{' '}
                                    <a href="#" style={{ color: linkClr, textDecoration: 'underline' }}>Terms of Service</a> and{' '}
                                    <a href="#" style={{ color: linkClr, textDecoration: 'underline' }}>Privacy Policy</a>.
                                </p>
                            </form>
                        </div>

                        {/* Sign in link */}
                        <p className="anim-fade-up d2" style={{ textAlign: 'center', fontSize: 13, color: textSub, marginTop: 20 }}>
                            Already have an account?{' '}
                            <button onClick={() => navigate('/login')} style={{ background: 'none', border: 'none', cursor: 'pointer', fontSize: 13, fontWeight: 600, color: linkClr, padding: 0 }}>
                                Sign in →
                            </button>
                        </p>
                    </div>
                </div>
            </div>
        </div>
    );
};

export default Register;
